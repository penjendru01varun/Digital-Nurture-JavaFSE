import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { EnrollmentService } from '../../services/enrollment.service';
import { Course } from '../../models/course.model';

@Component({
  selector: 'app-student-profile',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './student-profile.component.html',
  styleUrls: ['./student-profile.component.css']
})
export class StudentProfileComponent implements OnInit {
  enrolledCourses: Course[] = [];



  // Task 66: Inject EnrollmentService
  constructor(private enrollmentService: EnrollmentService) {}

  ngOnInit(): void {
    // Task 66: Load enrolled courses
    this.enrolledCourses = this.enrollmentService.getEnrolledCourses();
  }

  // Method to refresh profile data
  refreshProfile(): void {
    this.enrolledCourses = this.enrollmentService.getEnrolledCourses();
  }

  getGpa(): string {
    // Simple mock GPA calculation based on number of courses
    const count = this.enrolledCourses.length;
    if (count === 0) return 'N/A';
    if (count <= 2) return '3.5';
    if (count <= 4) return '3.2';
    return '3.0';
  }
}
