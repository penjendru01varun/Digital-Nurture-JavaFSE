import { Component } from '@angular/core';
import { CommonModule } from '@angular/common';

// Task 67: NotificationService - provided at component level
// This service is NOT provided at root level, so each NotificationComponent
// gets its own separate instance
class NotificationService {
  private notifications: string[] = [];

  addNotification(message: string): void {
    this.notifications.push(message);
  }

  getNotifications(): string[] {
    return this.notifications;
  }

  clear(): void {
    this.notifications = [];
  }
}

@Component({
  selector: 'app-notification',
  standalone: true,
  imports: [CommonModule],
  // Task 67: Component-level provider - creates a new instance per component
  providers: [NotificationService],
  templateUrl: './notification.component.html',
  styleUrls: ['./notification.component.css']
})
export class NotificationComponent {
  notifications: string[] = [];

  // COMMENT: Component-level providers create a new service instance
  // for this component and its children. This is useful when you need
  // isolated state per component instance, such as:
  // - Multiple form wizards with independent state
  // - Reusable components that need their own service instance
  // - Components that should not share state with other instances
  // Unlike root-level providers (singleton), component-level providers
  // create a new instance each time the component is instantiated.

  constructor(private notificationService: NotificationService) {
    // Load existing notifications
    this.notifications = this.notificationService.getNotifications();
  }

  addNotification(message: string): void {
    this.notificationService.addNotification(message);
    this.notifications = this.notificationService.getNotifications();
  }

  clearNotifications(): void {
    this.notificationService.clear();
    this.notifications = [];
  }
}

<!-- FILE: src/app/components/notification/notification.component.html -->

<div class="notification-component">
  <h4>Notifications</h4>
  
  <div class="notification-input">
    <input #messageInput placeholder="Enter notification..." (keyup.enter)="addNotification(messageInput.value); messageInput.value = ''">
    <button (click)="addNotification(messageInput.value); messageInput.value = ''">Add</button>
    <button (click)="clearNotifications()" class="btn-clear">Clear All</button>
  </div>
  
  <div class="notification-list">
    <div *ngIf="notifications.length === 0" class="no-notifications">
      No notifications
    </div>
    <div *ngFor="let notification of notifications; let i = index" class="notification-item">
      <span>{{ notification }}</span>
      <span class="notification-id">#{{ i + 1 }}</span>
    </div>
  </div>
  
  <p class="hint">This component has its own NotificationService instance.</p>
</div>
