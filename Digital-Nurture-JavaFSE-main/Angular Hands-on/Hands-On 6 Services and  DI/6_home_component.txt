import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { CourseService } from '../../services/course.service';
import { EnrollmentService } from '../../services/enrollment.service';

@Component({
  selector: 'app-home',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './home.component.html',
  styleUrls: ['./home.component.css']
})
export class HomeComponent implements OnInit {
  portalName = 'Student Course Portal';
  isPortalActive = true;
  message = '';
  searchTerm = '';
  totalCourses = 0;
  enrolledCount = 0;
  gpa = 3.8;

  // Task 61: Inject CourseService and EnrollmentService
  constructor(
    private courseService: CourseService,
    private enrollmentService: EnrollmentService
  ) {}

  ngOnInit(): void {
    // Task 61: Display live courses count
    this.totalCourses = this.courseService.getCourses().length;
    this.enrolledCount = this.enrollmentService.getEnrolledCourses().length;
    console.log('HomeComponent initialised — courses loaded');
  }

  onEnrollClick(): void {
    this.message = 'Enrollment opened!';
  }
}

<!-- FILE: src/app/pages/home/home.component.html -->

<h1>{{ portalName }}</h1>

<p>Welcome to the Student Course Portal. Manage your courses, track your progress, and enroll in new courses.</p>

<!-- Task 61: Stats row with live data -->
<div class="stats">
  <span>Courses Available: {{ totalCourses }}</span>
  <span>Enrolled: {{ enrolledCount }}</span>
  <span>GPA: {{ gpa }}</span>
</div>

<button [disabled]="!isPortalActive" (click)="onEnrollClick()">Enroll Now</button>
<p *ngIf="message">{{ message }}</p>

<input [(ngModel)]="searchTerm" placeholder="Search courses...">
<p>Searching for: {{ searchTerm }}</p>
