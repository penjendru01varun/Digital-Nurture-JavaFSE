// FILE: src/app/store/enrollment/enrollment.actions.ts

import { createAction, props } from '@ngrx/store';

// Task 99: Define enrollment actions
export const enrollInCourse = createAction(
  '[Enrollment] Enroll In Course',
  props<{ courseId: number }>()
);

export const unenrollFromCourse = createAction(
  '[Enrollment] Unenroll From Course',
  props<{ courseId: number }>()
);

export const setEnrolledCourses = createAction(
  '[Enrollment] Set Enrolled Courses',
  props<{ enrolledCourseIds: number[] }>()
);

export const loadEnrolledCourses = createAction(
  '[Enrollment] Load Enrolled Courses'
);

export const loadEnrolledCoursesSuccess = createAction(
  '[Enrollment] Load Enrolled Courses Success',
  props<{ enrolledCourseIds: number[] }>()
);

export const loadEnrolledCoursesFailure = createAction(
  '[Enrollment] Load Enrolled Courses Failure',
  props<{ error: string }>()
);
