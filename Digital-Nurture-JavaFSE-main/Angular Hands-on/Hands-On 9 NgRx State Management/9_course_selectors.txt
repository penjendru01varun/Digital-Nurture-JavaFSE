// FILE: src/app/store/course/course.selectors.ts

import { createFeatureSelector, createSelector } from '@ngrx/store';
import { CourseState } from './course.reducer';

// Task 95: Create feature selector
export const selectCourseState = createFeatureSelector<CourseState>('course');

// Task 95: Create selectors
export const selectAllCourses = createSelector(
  selectCourseState,
  (state: CourseState) => state.courses
);

export const selectCoursesLoading = createSelector(
  selectCourseState,
  (state: CourseState) => state.loading
);

export const selectCoursesError = createSelector(
  selectCourseState,
  (state: CourseState) => state.error
);

// Additional selectors
export const selectCourseById = (id: number) => createSelector(
  selectAllCourses,
  (courses) => courses.find(course => course.id === id)
);

export const selectCoursesByStatus = (status: string) => createSelector(
  selectAllCourses,
  (courses) => courses.filter(course => course.gradeStatus === status)
);

export const selectTotalCourses = createSelector(
  selectAllCourses,
  (courses) => courses.length
);
