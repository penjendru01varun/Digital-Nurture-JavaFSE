// FILE: src/app/store/course/course.reducer.ts

import { createReducer, on } from '@ngrx/store';
import { Course } from '../../models/course.model';
import * as CourseActions from './course.actions';

// Task 94: Define state interface
export interface CourseState {
  courses: Course[];
  loading: boolean;
  error: string | null;
}

// Initial state
export const initialState: CourseState = {
  courses: [],
  loading: false,
  error: null
};

// Task 94: Create reducer with createReducer and on handlers
export const courseReducer = createReducer(
  initialState,
  
  // Handle loadCourses
  on(CourseActions.loadCourses, (state): CourseState => ({
    ...state,
    loading: true,
    error: null
  })),
  
  // Handle loadCoursesSuccess
  on(CourseActions.loadCoursesSuccess, (state, { courses }): CourseState => ({
    ...state,
    courses,
    loading: false,
    error: null
  })),
  
  // Handle loadCoursesFailure
  on(CourseActions.loadCoursesFailure, (state, { error }): CourseState => ({
    ...state,
    loading: false,
    error
  })),
  
  // Handle addCourse
  on(CourseActions.addCourse, (state): CourseState => ({
    ...state,
    loading: true,
    error: null
  })),
  
  // Handle addCourseSuccess
  on(CourseActions.addCourseSuccess, (state, { course }): CourseState => ({
    ...state,
    courses: [...state.courses, course],
    loading: false
  })),
  
  // Handle updateCourse
  on(CourseActions.updateCourse, (state, { id, changes }): CourseState => ({
    ...state,
    courses: state.courses.map(course => 
      course.id === id ? { ...course, ...changes } : course
    )
  })),
  
  // Handle deleteCourse
  on(CourseActions.deleteCourse, (state, { id }): CourseState => ({
    ...state,
    courses: state.courses.filter(course => course.id !== id)
  }))
);
