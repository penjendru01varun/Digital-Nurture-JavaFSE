// FILE: src/app/app.config.ts

import { ApplicationConfig, provideZoneChangeDetection } from '@angular/core';
import { provideRouter } from '@angular/router';
import { provideHttpClient, withInterceptors } from '@angular/common/http';
import { routes } from './app.routes';
import { authInterceptor } from './interceptors/auth.interceptor';
import { errorHandlerInterceptor } from './interceptors/error-handler.interceptor';
import { loadingInterceptor } from './interceptors/loading.interceptor';

export const appConfig: ApplicationConfig = {
  providers: [
    provideZoneChangeDetection({ eventCoalescing: true }),
    provideRouter(routes),
    // Task 88: Register HTTP interceptors
    provideHttpClient(
      withInterceptors([
        authInterceptor,       // Order matters - auth first
        errorHandlerInterceptor, // Then error handling
        loadingInterceptor     // Finally loading
      ])
    )
  ]
};

/*
COMMENT: Interceptor Execution Order
- Request phase: Interceptors execute in registration order
  1. authInterceptor - Adds auth token
  2. errorHandlerInterceptor - Wraps for error handling
  3. loadingInterceptor - Shows loading indicator

- Response phase: Interceptors execute in reverse order
  3. loadingInterceptor - Hides loading indicator
  2. errorHandlerInterceptor - Catches errors
  1. authInterceptor - Response passes through auth (if needed)
*/
