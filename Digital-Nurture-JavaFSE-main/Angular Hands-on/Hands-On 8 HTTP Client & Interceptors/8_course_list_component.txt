import { Component, OnInit, OnDestroy } from '@angular/core';
import { CommonModule } from '@angular/common';
import { Router } from '@angular/router';
import { Subscription } from 'rxjs';
import { CourseCardComponent } from '../../components/course-card/course-card.component';
import { CourseService } from '../../services/course.service';
import { Course } from '../../models/course.model';

@Component({
  selector: 'app-course-list',
  standalone: true,
  imports: [CommonModule, CourseCardComponent],
  templateUrl: './course-list.component.html',
  styleUrls: ['./course-list.component.css']
})
export class CourseListComponent implements OnInit, OnDestroy {
  courses: Course[] = [];
  isLoading = true;
  errorMessage: string | null = null;
  private subscription: Subscription | null = null;

  constructor(
    private courseService: CourseService,
    private router: Router
  ) {}

  // Task 80: Subscribe to HTTP call
  ngOnInit(): void {
    this.loadCourses();
  }

  loadCourses(): void {
    this.isLoading = true;
    this.errorMessage = null;
    
    this.subscription = this.courseService.getCourses().subscribe({
      next: (courses) => {
        this.courses = courses;
        this.isLoading = false;
      },
      error: (err) => {
        this.errorMessage = err.message;
        this.isLoading = false;
        console.error('Error loading courses:', err);
      },
      complete: () => {
        // Task 80: Complete callback
        console.log('Course loading complete');
        this.isLoading = false;
      }
    });
  }

  // Task 70: Navigate to course detail
  onCourseClick(courseId: number): void {
    this.router.navigate(['/courses', courseId]);
  }

  // Task 71: Update search query parameter
  onSearch(searchTerm: string): void {
    this.router.navigate(['/courses'], {
      queryParams: { search: searchTerm || null }
    });
  }

  ngOnDestroy(): void {
    // Clean up subscription to prevent memory leaks
    if (this.subscription) {
      this.subscription.unsubscribe();
    }
  }

  trackByCourseId(index: number, course: Course): number {
    return course.id;
  }
}

<div class="course-list">
  <h2>Available Courses</h2>
  
  <!-- Search input with query params -->
  <div class="search-bar">

  <div *ngIf="isLoading; else courseGrid" class="loading">

  </div>
  
  <!-- Course grid -->
  <ng-template #courseGrid>
    <div *ngIf="courses.length > 0; else noCourses" class="course-grid">
      <!-- Task 70: Make cards clickable -->
      <div 
        *ngFor="let course of courses; trackBy: trackByCourseId" 
        class="course-card-wrapper"
        (click)="onCourseClick(course.id)">
        <app-course-card [course]="course"></app-course-card>
      </div>
    </div>
    
    <ng-template #noCourses>

    </ng-template>
  </ng-template>
</div>
