// FILE: src/app/services/auth.service.ts

import { Injectable } from '@angular/core';

@Injectable({
  providedIn: 'root'
})
export class AuthService {
  // Task 75: Hardcoded login state for demo
  private loggedIn = true; // Set to true for testing, false to test guard

  isLoggedIn(): boolean {
    return this.loggedIn;
  }

  login(): void {
    this.loggedIn = true;
  }

  logout(): void {
    this.loggedIn = false;
  }

  // Toggle login state (useful for testing)
  toggleLogin(): void {
    this.loggedIn = !this.loggedIn;
  }
}
