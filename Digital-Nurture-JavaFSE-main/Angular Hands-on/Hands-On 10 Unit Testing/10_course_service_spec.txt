// FILE: src/app/services/course.service.spec.ts

import { TestBed } from '@angular/core/testing';
import { HttpClientTestingModule, HttpTestingController } from '@angular/common/http/testing';
import { CourseService } from './course.service';
import { Course } from '../models/course.model';

describe('CourseService', () => {
  let service: CourseService;
  let httpMock: HttpTestingController;
  const apiUrl = 'http://localhost:3000/courses';

  // Task 106: Configure TestBed with HttpClientTestingModule
  beforeEach(() => {
    TestBed.configureTestingModule({
      imports: [HttpClientTestingModule],
      providers: [CourseService]
    });

    service = TestBed.inject(CourseService);
    httpMock = TestBed.inject(HttpTestingController);
  });

  afterEach(() => {
    // Task 107: Verify no outstanding HTTP requests
    httpMock.verify();
  });

  // Task 107: Test getCourses()
  it('should retrieve courses from API', () => {
    const mockCourses: Course[] = [
      { id: 1, name: 'Data Structures', code: 'CS101', credits: 3, gradeStatus: 'passed' },
      { id: 2, name: 'Algorithms', code: 'CS201', credits: 4, gradeStatus: 'pending' }
    ];

    service.getCourses().subscribe(courses => {
      expect(courses.length).toBe(2);
      expect(courses).toEqual(mockCourses);
    });

    // Expect one HTTP request to the correct URL
    const req = httpMock.expectOne(apiUrl);
    expect(req.request.method).toBe('GET');
    req.flush(mockCourses);
  });

  // Test getCourseById()
  it('should retrieve a single course by ID', () => {
    const mockCourse: Course = { 
      id: 1, 
      name: 'Data Structures', 
      code: 'CS101', 
      credits: 3, 
      gradeStatus: 'passed' 
    };

    service.getCourseById(1).subscribe(course => {
      expect(course).toEqual(mockCourse);
    });

    const req = httpMock.expectOne(`${apiUrl}/1`);
    expect(req.request.method).toBe('GET');
    req.flush(mockCourse);
  });

  // Test createCourse()
  it('should create a new course', () => {
    const newCourse: Omit<Course, 'id'> = {
      name: 'Web Development',
      code: 'CS401',
      credits: 3,
      gradeStatus: 'pending'
    };

    const createdCourse: Course = {
      ...newCourse,
      id: 4
    };

    service.createCourse(newCourse).subscribe(course => {
      expect(course).toEqual(createdCourse);
    });

    const req = httpMock.expectOne(apiUrl);
    expect(req.request.method).toBe('POST');
    expect(req.request.body).toEqual(newCourse);
    req.flush(createdCourse);
  });

  // Test updateCourse()
  it('should update an existing course', () => {
    const updates = { name: 'Updated Course' };
    const updatedCourse: Course = {
      id: 1,
      name: 'Updated Course',
      code: 'CS101',
      credits: 3,
      gradeStatus: 'passed'
    };

    service.updateCourse(1, updates).subscribe(course => {
      expect(course).toEqual(updatedCourse);
    });

    const req = httpMock.expectOne(`${apiUrl}/1`);
    expect(req.request.method).toBe('PUT');
    expect(req.request.body).toEqual(updates);
    req.flush(updatedCourse);
  });

  // Test deleteCourse()
  it('should delete a course', () => {
    service.deleteCourse(1).subscribe(() => {
      expect(true).toBeTruthy();
    });

    const req = httpMock.expectOne(`${apiUrl}/1`);
    expect(req.request.method).toBe('DELETE');
    req.flush(null);
  });

  // Task 108: Test error handling
  it('should handle errors when API fails', () => {
    const errorMessage = 'Failed to load courses. Please try again.';

    service.getCourses().subscribe({
      next: () => fail('Expected error, got success'),
      error: (error) => {
        expect(error.message).toBe('Failed to load courses. Please try again.');
      }
    });

    const req = httpMock.expectOne(apiUrl);
    req.flush('Error loading courses', {
      status: 500,
      statusText: 'Internal Server Error'
    });
  });

  // Test with error status 404
  it('should handle 404 errors', () => {
    service.getCourseById(999).subscribe({
      error: (error) => {
        expect(error.message).toBe('Failed to load courses. Please try again.');
      }
    });

    const req = httpMock.expectOne(`${apiUrl}/999`);
    req.flush('Course not found', {
      status: 404,
      statusText: 'Not Found'
    });
  });
});
