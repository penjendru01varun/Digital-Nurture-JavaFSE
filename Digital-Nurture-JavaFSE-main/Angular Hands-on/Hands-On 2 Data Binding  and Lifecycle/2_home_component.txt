import { Component, OnInit, OnDestroy } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'app-home',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './home.component.html',
  styleUrls: ['./home.component.css']
})
export class HomeComponent implements OnInit, OnDestroy {
  // Task 11: String interpolation property
  portalName = 'Student Course Portal';
  
  // Task 12: Property binding property
  isPortalActive = true;
  
  // Task 13: Event binding property
  message = '';
  
  // Task 14: Two-way binding property
  searchTerm = '';
  
  // Task 16: Lifecycle hook - ngOnInit
  ngOnInit(): void {
    // Simulating fetch of course count
    console.log('HomeComponent initialised — courses loaded');
  }
  
  // Task 13: Event binding method
  onEnrollClick(): void {
    this.message = 'Enrollment opened!';
  }
  
  // Task 17: Lifecycle hook - ngOnDestroy
  ngOnDestroy(): void {
    console.log('HomeComponent destroyed');
  }
}

<!-- Task 15: Code comment explaining the difference -->
<!-- 
  [property] = One-way binding: Data flows from component → DOM
  Example: [disabled]="!isPortalActive" binds the component property to the DOM element's disabled attribute
  
  [(ngModel)] = Two-way binding: Data flows DOM ↔ component (bidirectional)
  Example: [(ngModel)]="searchTerm" - any change in input updates component property,
  and any change in component property updates the input value
  It's shorthand for: [ngModel]="searchTerm" (ngModelChange)="searchTerm = $event"
-->
