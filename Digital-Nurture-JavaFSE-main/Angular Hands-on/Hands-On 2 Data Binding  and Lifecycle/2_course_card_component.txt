import { Component, Input, Output, EventEmitter, OnChanges, SimpleChanges } from '@angular/core';
import { CommonModule } from '@angular/common';

@Component({
  selector: 'app-course-card',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './course-card.component.html',
  styleUrls: ['./course-card.component.css']
})
export class CourseCardComponent implements OnChanges {
  // Task 18 & 20: @Input with proper type definition
  @Input() course!: {
    id: number;
    name: string;
    code: string;
    credits: number;
  };
  
  // Task 21: @Output EventEmitter with payload type
  @Output() enrollRequested = new EventEmitter<number>();
  
  // Task 18: ngOnChanges lifecycle hook
  ngOnChanges(changes: SimpleChanges): void {
    if (changes['course']) {
      const previous = changes['course'].previousValue;
      const current = changes['course'].currentValue;
      console.log('Course changed - Previous:', previous, 'Current:', current);
    }
  }
  
  // Task 21: Method to emit enrollment event
  onEnroll(): void {
    this.enrollRequested.emit(this.course.id);
  }
}

  <p><strong>Credits:</strong> {{ course.credits }}</p>
  <p><strong>ID:</strong> {{ course.id }}</p>
  
  <!-- Task 21: Enroll button with event emission -->
  <button (click)="onEnroll()">Enroll</button>
