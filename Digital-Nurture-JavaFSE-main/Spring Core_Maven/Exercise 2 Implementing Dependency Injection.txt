<beans xmlns="http://www.springframework.org/schema/beans"

    <!-- Repository Bean -->
    <bean id="bookRepository" class="com.library.repository.BookRepository">
        <property name="databaseName" value="LibraryDB"/>
    </bean>

    <!-- Service Bean with Setter Injection -->
    <bean id="bookService" class="com.library.service.BookService">
        <property name="bookRepository" ref="bookRepository"/>
    </bean>

</beans>


 2. Updated BookRepository.java


package com.library.repository;

import java.util.ArrayList;
import java.util.List;

public class BookRepository {
    private String databaseName;
    private List<String> books = new ArrayList<>();

    public BookRepository() {
        books.add("Spring Framework Essentials");
        books.add("Java Design Patterns");
        books.add("Clean Code");
    }

    public void setDatabaseName(String databaseName) {
        this.databaseName = databaseName;
    }

    public String getDatabaseName() {
        return databaseName;
    }

    public List<String> getAllBooks() {
        return new ArrayList<>(books);
    }

    public void addBook(String book) {
        books.add(book);
    }

    public boolean removeBook(String book) {
        return books.remove(book);
    }

    public int getBookCount() {
        return books.size();
    }
}

 3. Updated BookService.java (Setter for DI)


package com.library.service;

import com.library.repository.BookRepository;
import java.util.List;

public class BookService {
    private BookRepository bookRepository;

    public BookService() {
    }

    public void setBookRepository(BookRepository bookRepository) {
        this.bookRepository = bookRepository;
    }

    public BookRepository getBookRepository() {
        return bookRepository;
    }

    public List<String> listBooks() {
        return bookRepository.getAllBooks();
    }

    public void addNewBook(String book) {
        bookRepository.addBook(book);
    }

    public boolean deleteBook(String book) {
        return bookRepository.removeBook(book);
    }

    public int getTotalBooks() {
        return bookRepository.getBookCount();
    }

    public String getDatabaseInfo() {
        return "Connected to: " + bookRepository.getDatabaseName();
    }
}
 4. LibraryManagementApplication.java (Test Class)


package com.library;

import com.library.service.BookService;
import org.springframework.context.ApplicationContext;
import org.springframework.context.support.ClassPathXmlApplicationContext;

public class LibraryManagementApplication {
    public static void main(String[] args) {
        ApplicationContext context = new ClassPathXmlApplicationContext("applicationContext.xml");

        BookService bookService = (BookService) context.getBean("bookService");

        System.out.println("=== Dependency Injection Verification ===");
        System.out.println("Database Info: " + bookService.getDatabaseInfo());
        System.out.println("Repository injected: " + (bookService.getBookRepository() != null));
        System.out.println();

        System.out.println("=== Initial Book List ===");
        bookService.listBooks().forEach(System.out::println);
        System.out.println("Total books: " + bookService.getTotalBooks());
        System.out.println();

        System.out.println("=== Adding New Book ===");
        bookService.addNewBook("Microservices Architecture");
        bookService.listBooks().forEach(System.out::println);
        System.out.println("Total books: " + bookService.getTotalBooks());
        System.out.println();

        System.out.println("=== Deleting Book ===");
        bookService.deleteBook("Clean Code");
        bookService.listBooks().forEach(System.out::println);
        System.out.println("Total books: " + bookService.getTotalBooks());
    }
}

public class BookService {
