import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

interface ExternalApi {
    void sendData(String data);
    void processUser(int userId, String username);
    String getData();
}

class MyService {
    private final ExternalApi externalApi;

    public MyService(ExternalApi externalApi) {
        this.externalApi = externalApi;
    }

    public void doWork() {
        externalApi.sendData("important");
    }

    public void registerUser(int id, String name) {
        externalApi.processUser(id, name);
    }

    public String fetchData() {
        return externalApi.getData();
    }
}

public class MyServiceTest {

    @Test
    public void testVerifySendDataCalled() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        service.doWork();

        verify(mockApi).sendData("important");
    }

    @Test
    public void testVerifyProcessUserWithExactArgs() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        service.registerUser(42, "Alice");

        verify(mockApi).processUser(42, "Alice");
    }

    @Test
    public void testVerifyNeverCalled() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        verify(mockApi, never()).sendData(anyString());
    }

    @Test
    public void testVerifyNumberOfCalls() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        service.doWork();
        service.doWork();

        verify(mockApi, times(2)).sendData("important");
    }

    @Test
    public void testVerifyAtLeastOnce() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        service.fetchData();
        service.fetchData();

        verify(mockApi, atLeast(1)).getData();
    }

    @Test
    public void testVerifyWithArgumentMatchers() {
        ExternalApi mockApi = Mockito.mock(ExternalApi.class);
        MyService service = new MyService(mockApi);

        service.registerUser(99, "Bob");

        verify(mockApi).processUser(anyInt(), eq("Bob"));
    }
}
