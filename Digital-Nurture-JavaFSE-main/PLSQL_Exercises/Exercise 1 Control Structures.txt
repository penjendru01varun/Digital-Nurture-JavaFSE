 Schema Setup

CREATE TABLE Customers (
    CustomerID NUMBER PRIMARY KEY,
    Name VARCHAR2(100),
    DOB DATE,
    Balance NUMBER,
    LastModified DATE
);

CREATE TABLE Accounts (
    AccountID NUMBER PRIMARY KEY,
    CustomerID NUMBER,
    AccountType VARCHAR2(20),
    Balance NUMBER,
    LastModified DATE,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID)
);

CREATE TABLE Transactions (
    TransactionID NUMBER PRIMARY KEY,
    AccountID NUMBER,
    TransactionDate DATE,
    Amount NUMBER,
    TransactionType VARCHAR2(10),
    FOREIGN KEY (AccountID) REFERENCES Accounts(AccountID)
);

CREATE TABLE Loans (
    LoanID NUMBER PRIMARY KEY,
    CustomerID NUMBER,
    LoanAmount NUMBER,
    InterestRate NUMBER,
    StartDate DATE,
    EndDate DATE,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID)
);

CREATE TABLE Employees (
    EmployeeID NUMBER PRIMARY KEY,
    Name VARCHAR2(100),
    Position VARCHAR2(50),
    Salary NUMBER,
    Department VARCHAR2(50),
    HireDate DATE
);

ALTER TABLE Customers ADD IsVIP VARCHAR2(5);

 Scenario 1: Senior Citizen Discount

DECLARE
    CURSOR c_customers IS
        SELECT CustomerID, Name, DOB, Balance
        FROM Customers;
    
    v_customer_id Customers.CustomerID%TYPE;
    v_name Customers.Name%TYPE;
    v_dob Customers.DOB%TYPE;
    v_age NUMBER;
    v_balance Customers.Balance%TYPE;
    v_loan_count NUMBER;
BEGIN
    OPEN c_customers;
    LOOP
        FETCH c_customers INTO v_customer_id, v_name, v_dob, v_balance;
        EXIT WHEN c_customers%NOTFOUND;
        
        v_age := FLOOR(MONTHS_BETWEEN(SYSDATE, v_dob) / 12);
        
        IF v_age > 60 THEN
            SELECT COUNT(*) INTO v_loan_count
            FROM Loans
            WHERE CustomerID = v_customer_id;
            
            IF v_loan_count > 0 THEN
                UPDATE Loans
                SET InterestRate = InterestRate - 1
                WHERE CustomerID = v_customer_id;
                
                DBMS_OUTPUT.PUT_LINE('Customer: ' || v_name || ' (Age: ' || v_age || ') - 1% discount applied to all loans.');
            ELSE
                DBMS_OUTPUT.PUT_LINE('Customer: ' || v_name || ' (Age: ' || v_age || ') - No loans found.');
            END IF;
        END IF;
    END LOOP;
    CLOSE c_customers;
    
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Senior citizen discount processing completed.');
END;

Scenario 2: VIP Status Update


DECLARE
    CURSOR c_customers IS
        SELECT CustomerID, Name, Balance, IsVIP
        FROM Customers;
    
    v_customer_id Customers.CustomerID%TYPE;
    v_name Customers.Name%TYPE;
    v_balance Customers.Balance%TYPE;
    v_is_vip Customers.IsVIP%TYPE;
    v_updated_count NUMBER := 0;
BEGIN
    OPEN c_customers;
    LOOP
        FETCH c_customers INTO v_customer_id, v_name, v_balance, v_is_vip;
        EXIT WHEN c_customers%NOTFOUND;
        
        IF v_balance > 10000 THEN
            UPDATE Customers
            SET IsVIP = 'TRUE'
            WHERE CustomerID = v_customer_id;
            
            v_updated_count := v_updated_count + 1;
            DBMS_OUTPUT.PUT_LINE('Customer: ' || v_name || ' (Balance: $' || v_balance || ') - Promoted to VIP.');
        ELSE
            UPDATE Customers
            SET IsVIP = 'FALSE'
            WHERE CustomerID = v_customer_id;
            
            DBMS_OUTPUT.PUT_LINE('Customer: ' || v_name || ' (Balance: $' || v_balance || ') - VIP status removed.');
        END IF;
    END LOOP;
    CLOSE c_customers;
    
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('VIP status update completed. Total VIP customers: ' || v_updated_count);
END;
/

Scenario 3: Loan Due Reminder


DECLARE
    CURSOR c_loans_due IS
        SELECT l.LoanID, c.CustomerID, c.Name, c.Balance, l.LoanAmount, l.InterestRate, l.EndDate
        FROM Loans l
        JOIN Customers c ON l.CustomerID = c.CustomerID
        WHERE l.EndDate BETWEEN SYSDATE AND SYSDATE + 30;
    
    v_loan_id Loans.LoanID%TYPE;
    v_customer_id Customers.CustomerID%TYPE;
    v_name Customers.Name%TYPE;
    v_balance Customers.Balance%TYPE;
    v_loan_amount Loans.LoanAmount%TYPE;
    v_interest_rate Loans.InterestRate%TYPE;
    v_end_date Loans.EndDate%TYPE;
    v_days_remaining NUMBER;
    v_total_due NUMBER;
BEGIN
    OPEN c_loans_due;
    LOOP
        FETCH c_loans_due INTO v_loan_id, v_customer_id, v_name, v_balance, v_loan_amount, v_interest_rate, v_end_date;
        EXIT WHEN c_loans_due%NOTFOUND;
        
        v_days_remaining := TRUNC(v_end_date - SYSDATE);
        v_total_due := v_loan_amount + (v_loan_amount * v_interest_rate / 100);
        
        DBMS_OUTPUT.PUT_LINE('============================================');
        DBMS_OUTPUT.PUT_LINE('REMINDER: Loan Due Soon');
        DBMS_OUTPUT.PUT_LINE('============================================');
        DBMS_OUTPUT.PUT_LINE('Customer Name: ' || v_name);
        DBMS_OUTPUT.PUT_LINE('Customer ID: ' || v_customer_id);
        DBMS_OUTPUT.PUT_LINE('Loan ID: ' || v_loan_id);
        DBMS_OUTPUT.PUT_LINE('Loan Amount: $' || v_loan_amount);
        DBMS_OUTPUT.PUT_LINE('Interest Rate: ' || v_interest_rate || '%');
        DBMS_OUTPUT.PUT_LINE('Total Amount Due: $' || v_total_due);
        DBMS_OUTPUT.PUT_LINE('Due Date: ' || TO_CHAR(v_end_date, 'DD-MON-YYYY'));
        DBMS_OUTPUT.PUT_LINE('Days Remaining: ' || v_days_remaining);
        DBMS_OUTPUT.PUT_LINE('Current Balance: $' || v_balance);
        DBMS_OUTPUT.PUT_LINE('--------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('Please ensure timely payment to avoid penalties.');
        DBMS_OUTPUT.PUT_LINE('Contact our support team for assistance.');
        DBMS_OUTPUT.PUT_LINE('============================================');
        DBMS_OUTPUT.PUT_LINE('');
    END LOOP;
    CLOSE c_loans_due;
    
    IF c_loans_due%ROWCOUNT = 0 THEN
        DBMS_OUTPUT.PUT_LINE('No loans due within the next 30 days.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Loan reminder processing completed. Total reminders sent: ' || c_loans_due%ROWCOUNT);
    END IF;
END;
